% ===== Author's machine: the vertical chain =====
\node[tool] (ocr) {\texttt{sms\_transcribe.py}\\\scriptsize OCR, hand check, images deleted};
\node[tool,below=5mm of ocr] (collect) {\texttt{sms\_collect.py}\\\scriptsize validates each row, ingests};
\node[tool,below=5mm of collect] (annot) {\texttt{sms\_annotate.py}\\\scriptsize two blind annotators, adjudication};
\node[tool,below=5mm of annot] (tmpl) {\texttt{sms\_templates.py}\\\scriptsize word 4-shingles, Jaccard $\tau=0.8$};
\node[tool,below=5mm of tmpl] (split) {\texttt{sms\_split.py}\\\scriptsize 70/15/15 per template, seed 1602};
\node[tool,below=5mm of split] (pub) {\texttt{sms\_publish.py}\\\scriptsize schema projection, rules checked};
\node[tool,right=14mm of annot] (imp) {\texttt{sms\_import.py}\\\scriptsize tokens remapped, 310 rows dropped};
\node[fit=(ocr)(pub)(imp),draw=fgMuted,densely dotted,inner sep=7pt] (auth) {};
\node[head,anchor=south west,fill=white,inner sep=1pt] at ([yshift=2.5mm]auth.north west) {Author's machine};

\draw[arr] (ocr) -- (collect);

\draw[arr] (tmpl) -- (split);
\draw[arr] (split) -- (pub);

\node[note,anchor=west] at ([xshift=2mm,yshift=-7mm]imp.south west) {held out: \texttt{split=external},\\never annotated, label kept};

% ===== Contributor's device =====

\node[src,above=5mm of paste] (shot) {screenshots\\\scriptsize cropped, own details blacked out};
\node[fit=(shot)(paste)(redact),draw=fgSource!60,densely dotted,inner sep=7pt] (dev) {};

\draw[arr] (paste) -- (redact);
